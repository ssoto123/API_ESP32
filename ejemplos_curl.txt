# Leer los sensores (GET)
curl http://192.168.1.50/api/sensores

# Encender el LED (POST con cuerpo JSON)
curl -X POST http://192.168.1.50/api/led \
  -H "Content-Type: application/json" \
  -d "{\"led\":true}"

# Apagarlo
curl -X POST http://192.168.1.50/api/led \
  -H "Content-Type: application/json" \
  -d "{\"led\":false}"

# Ver tambien los encabezados y el codigo de estado
curl -i http://192.168.1.50/api/noexiste
